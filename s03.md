# s03: peemission

在tool執行之前, 決定是否執行該指令

問題:

1. s02中, 檔案系統方面有安全系統(`safe_path()`), **但bash沒有, 隨時都有可能執行不可逆的破壞指令**

s03分成了三個步驟: 
**1. 會直接ban掉的危險指令**

**2. 設立規則, 透過上下文來判斷指令為危險指令還是可疑指令**

**3. 符合規則的可疑指令會透過詢問使用者的方式來決定是否執行**

核心程式碼: 
```
# Gate 1: Hard deny list - always forbidden
DENY_LIST = ["rm -rf /", "sudo", "shutdown", "reboot", "mkfs", "dd if=", "> /dev/sda"] #會直接被禁止掉的危險指令

def check_deny_list(command: str) -> str | None: # 尋找是否有危險指令, 有則直接ban掉
    for pattern in DENY_LIST:
        if pattern in command:
            return f"Blocked: '{pattern}' is on the deny list"
    return None

# Gate 2: Rule matching - context-dependent checks 
#設立規則, 符合條件的指令就會詢問使用者是否仍要執行
DESTRUCTIVE_COMMAND_WORD = re.compile(
    r"(?i)(?:^|[;&|()\n])\s*(?:rm|del)(?=\s|$|[;&|()])"
)
#(?i)               => 不論大小寫: RM -rf test.txt / Rm -rf test.txt
#(?:^|[;&|()\n])    => 必須再句首或分隔號後: rm file.txt / echo hello; rm secret.txt
#\s*                => 允許零個或多個空白字元: rm     file.txt / rm-rf
#(?:rm|del)         => rm 或 del: rm old_code.py / del old_code.py
#(?=\s|$|[;&|()])   => 後面必須接空白、結尾或分隔符號: python -m firmware_tool不會被阻止

查找指令是否符合
def contains_destructive_command(command: str) -> bool:
    return bool(DESTRUCTIVE_COMMAND_WORD.search(command))


PERMISSION_RULES = [
    
    #防止ai移動路徑, 只要路徑更改了就直接報錯
    {"tools": ["read_file", "write_file", "edit_file"],
     "check": lambda args: not (WORKDIR / args.get("path", "")).resolve().is_relative_to(WORKDIR), #防止ai偷看或修改內容
     "message": "Writing outside workspace"},
    
    #可疑指令判斷
    {"tools": ["bash"],
     "check": lambda args: contains_destructive_command(args.get("command", "")) or
     any(kw in args.get("command", "") for kw in ["rm ", "> /etc/", "chmod 777"]),
     "message": "Potentially destructive command"},
]

#發送可疑指令的原因
def check_rules(tool_name: str, args: dict) -> str | None:
    for rule in PERMISSION_RULES:
        if tool_name in rule["tools"] and rule["check"](args): # 如違反規則則阻止並回傳理由
            return rule["message"]
    return None

# Gate 3: User approval - wait for confirmation after rule match
def ask_user(tool_name: str, args: dict, reason: str) -> str: #給使用者自行判斷是否繼續執行可疑指令
    print(f"\n\033[33m[permission] {reason}\033[0m") #印出有什麼可疑訊息
    print(f"   Tool: {tool_name}({args})")
    choice = input("   Allow? [y/N] ").strip().lower()
    return "allow" if choice in ("y", "yes") else "deny" # 讓使用者決定是否執行


# Pipeline: all three gates chained
def check_permission(block) -> bool:
    if block.name == "bash":
        reason = check_deny_list(block.input.get("command", "")) # 如果出現rm等禁止指令, 則直接ban掉不給使用者考慮的機會
        if reason:
            print(f"\n\033[31m[blocked] {reason}\033[0m")
            return False
    reason = check_rules(block.name, block.input) # 如果觸發風險指令, 則問使用者是否繼續執行
    if reason:
        decision = ask_user(block.name, block.input, reason)
        if decision == "deny":
            return False
    return True
```
